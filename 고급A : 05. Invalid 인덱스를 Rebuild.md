● 1. INVALID 객체

-- 테이블 생성
CREATE TABLE emp_test 
( emp_id   NUMBER
, emp_name VARCHAR2(50) );

-- 프로시저 생성
CREATE OR REPLACE PROCEDURE proc_emp_test
    AS
    v_name VARCHAR2(50);
BEGIN
    SELECT emp_name INTO v_name
      FROM emp_test
     WHERE emp_id = 1;
    DBMS_OUTPUT.PUT_LINE('이름: ' || v_name);
END;
/


-- 현재 상태 확인
SELECT object_name, object_type, status
  FROM user_objects
 WHERE object_name = 'PROC_EMP_TEST';
→ VALID


-- 컬럼 변경  →  현재 상태 확인 시 INVALID로 변함
ALTER TABLE emp_test DROP COLUMN emp_name;
→ INVALID

ALTER PROCEDURE proc_emp_test COMPILE;           -- 컴파일
EXEC proc_emp_test;                              -- 컴파일 해도 실패

ALTER TABLE emp_test ADD emp_name VARCHAR2(50);  -- 복구
ALTER PROCEDURE proc_emp_test COMPILE;           -- 다시 컴파일
  
-- 다시 현재 상태 확인
SELECT object_name, object_type, status
  FROM user_objects
 WHERE object_name = 'PROC_EMP_TEST';
→ VALID

● 2. UNUSABLE 인덱스 

-- 테이블 생성
CREATE TABLE order_test
( order_id   NUMBER
, order_date DATE );

-- 인덱스 생성
CREATE INDEX idx_order_date ON order_test(order_date);

-- 인덱스 현재 상태 확인
SELECT index_name, status
  FROM user_indexes
 WHERE index_name = 'IDX_ORDER_DATE';
→ VALID

-- 다른 테이블 스페이스로 MOVE  →  현재 인덱스 상태 확인 시 UNUSABLE로 변함
ALTER TABLE order_test MOVE TABLESPACE ts300;
→ UNUSABLE

-- 인덱스 재구축
ALTER INDEX idx_order_date REBUILD ONLINE;

-- 인덱스 현재 상태 다시 확인
SELECT index_name, status
  FROM user_indexes
 WHERE index_name = 'IDX_ORDER_DATE';
→ VALID
