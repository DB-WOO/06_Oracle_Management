# 테이블 생성쿼리 추출 

SET LONG 2000000        -- 출력할 데이터의 길이를 늘림 (기본값이 매우 작음)
SET PAGESIZE 0          -- 페이지 구분선과 제목을 없앰
SET LINESIZE 200        -- 한 줄에 표시될 글자 수를 늘림
SET TRIMSPOOL ON        -- 줄 끝의 공백 제거

-- EMPLOYEES 테이블 생성문 추출
SELECT DBMS_METADATA.GET_DDL('TABLE', 'EMPLOYEES', 'HR') FROM DUAL;   → HR을 전부 HR2로 수정

   connect to hr2
   identified by hr2
   using '[유저 서비스명]';


select object_type, count(*)
  from all_objects
 where owner='HR'
 group by object_type
 order by 1 asc;
 
select object_type, count(*)
  from all_objects@hr2_link
 where owner='HR2'
 group by object_type
