● 2. 테이블 스페이스 종류

① Permanent 영구 테이블 스페이스 : 사용자 데이터와 데이터베이스 객체를 영구적으로 저장하는 테이블스페이스
   테이블, 인덱스, 뷰, 프로시저 - 실제 객체 저장
   SYSTEM, SYSAUX, USERS .....
   Set as default permanent tablespace 체크 : 기본 테이블 스페이스로 지
  
② Temporary 임시 테이블 스페이스 : SQL 처리 중 임시 작업 공간으로 사용되는 테이블스페이스
   정렬, 해시 조인, 임시 결과셋 등 메모리 부족 시 디스크로 내려쓰는 공간
   세션 종료 시 데이터 자동 삭제
   성능에 직접 영향을 미치므로 대용량 정렬 쿼리 환경에서 크기 관리 중요
   Set as default temporary tablespace 체크 : DB 전체 기본 임시 테이블스페이스로 지정

③ Undo 테이블 스페이스 : 트랜잭션 롤백과 읽기 일관성을 위한 이전 데이터를 저장하는 테이블스페이스
   DML 수행 전에 변경 전 데이터 저장
   ROLLBACK시 이 데이터를 이용해 원상복구
   Flashback Query, 읽기 일관성 보장에도 활용
   Undo Retention Guarantee: Yes 설정 시 지정 시간 동안 Undo 데이터 강제 보존 (Flashback 활용 시 중요)
   Oracle 9i 이후 자동 관리 방식 (AUM) : 기본값 UNDOTBS1


구분         저장 내용                 영속성                     대표 예
Permanent    테이블, 인덱스 등 객체    영구                       USERS, SYSTEM
Temporary    정렬, 해시 작업 공간      임시 (세션 종료 시 삭제)   TEMP
Undo         변경 전 데이터            보존 기간 내               UNDOTBS1




- Permanent 테이블 스페이스

select tablespace_name, status, contents, extent_management, segment_space_management
  from dba_tablespaces;
TABLESPACE_NAME                STATUS    CONTENTS              EXTENT_MAN SEGMEN
------------------------------ --------- --------------------- ---------- ------
SYSTEM                         ONLINE    PERMANENT             LOCAL      MANUAL
SYSAUX                         ONLINE    PERMANENT             LOCAL      AUTO
UNDOTBS1                       ONLINE    UNDO                  LOCAL      MANUAL
TEMP                           ONLINE    TEMPORARY             LOCAL      MANUAL
USERS                          ONLINE    PERMANENT             LOCAL      AUTO
TS01                           ONLINE    PERMANENT             LOCAL      AUTO
TS100                          ONLINE    PERMANENT             LOCAL      AUTO
TS600                          ONLINE    PERMANENT             LOCAL      AUTO
TS7000                         ONLINE    PERMANENT             LOCAL      AUTO


select *                          -- 기본 Permanent 테이블 스페이스 확인
  from database_properties
 where property_name='DEFAULT_PERMANENT_TABLESPACE';
  
PROPERTY_NAME                 PROPERTY_VALUE DESCRIPTION
----------------------------- -------------- ------------------------------------
DEFAULT_PERMANENT_TABLESPACE  USERS          Name of default permanent tablespace


select username, default_tablespace  -- SCOTT  TS01  -- 테이블스페이스 명시 안하고 테이블 만들면
  from dba_users                                     -- TS01에 생성됨
 where username='SCOTT';
  
alter  database default tablespace users;  -- default permanent 테이블 스페이스 변경






- Temp 테이블 스페이스 

select *
  from database_properties
 where property_name='DEFAULT_TEMP_TABLESPACE';

PROPERTY_NAME             PROPERTY_VALUE   DESCRIPTION
------------------------- ---------------- ------------------------------------
DEFAULT_TEMP_TABLESPACE   TEMP             Name of default temporary tablespace


select tablespace_name, contents
  from dba_tablespaces
 where contents = 'TEMPORARY';
  
TABLESPACE_NAME                CONTENTS
------------------------------ ---------------------
TEMP                           TEMPORARY 

  

  (생성 : 문법 살짝 다름)
create temporary tablespace temp3 tempfile '[ORACLE_BASE]/oradata/[ORACLE_SID]/temp03a.dbf' size 2m;   -- 일부터 작게 만듬

alter database default temporary tablespace temp3;

select username, default_tablespace, temporary_tablespace  -- temp3로 바뀜
  from dba_users
 where username='SCOTT';
