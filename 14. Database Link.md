# 테이블 생성쿼리 추출 

SET LONG 2000000        -- 출력할 데이터의 길이를 늘림 (기본값이 매우 작음)
SET PAGESIZE 0          -- 페이지 구분선과 제목을 없앰
SET LINESIZE 200        -- 한 줄에 표시될 글자 수를 늘림
SET TRIMSPOOL ON        -- 줄 끝의 공백 제거

-- EMPLOYEES 테이블 생성문 추출
SELECT DBMS_METADATA.GET_DDL('TABLE', 'EMPLOYEES', 'HR') FROM DUAL;   → HR을 전부 HR2로 수정

# 유저 인스턴스로 접속 후 링크 생성 + 테이블 스페이스 생성

connect hr2/hr2

create database link hr_link
     connect to hr
     identified by hr
     using '[Oracle 서비스명]'; 


※ 위에서 생성한 EMPLOYEES 테이블 생성문 추출 쿼리로 테이블 생성 후 이관

insert into employees
select * from employees@hr_link;
commit;
